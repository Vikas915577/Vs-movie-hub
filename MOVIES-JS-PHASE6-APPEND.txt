/* VS Movie Hub · Phase 6 loader. Append ONCE after the closing ]; in movies.js.
   Keep all 20 movie records unchanged. Do NOT add the Phase 5 Supabase loader.
*/
(() => {
  if (window.__VS_PHASE6_LOADER__) return;
  window.__VS_PHASE6_LOADER__ = true;
  const source = document.currentScript && document.currentScript.src
    ? document.currentScript.src
    : window.location.href;
  const loader = document.createElement('script');
  loader.src = new URL('./phase6-loader.js', source).href;
  loader.async = true;
  loader.onerror = () => console.warn('VS Movie Hub Phase 6 loader could not load.');
  document.head.appendChild(loader);
})();
